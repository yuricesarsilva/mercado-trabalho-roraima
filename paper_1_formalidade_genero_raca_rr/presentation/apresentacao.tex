\section{Introdu\c{c}\~ao}
% =============================================================================

\begin{frame}{O problema}
  \begin{itemize}
    \item Roraima tem um mercado de trabalho pequeno, com forte presen\c{c}a de informalidade, dependência do setor p\'ublico e composi\c{c}\~ao demogr\'afica marcada por fluxo migrat\'orio (fronteira com a Venezuela).
    \item Diferenciais de rendimento por formalidade, g\^enero e ra\c{c}a/cor s\~ao amplamente documentados para o Brasil como um todo, mas pouco explorados especificamente para Roraima.
    \item \highlight{Pergunta central:} mantidas constantes ocupa\c{c}\~ao, atividade econ\^omica, escolaridade, idade e per\'iodo, quais diferenciais de rendimento persistem por formalidade, g\^enero e ra\c{c}a/cor no mercado de trabalho de Roraima -- e como eles se relacionam entre si?
  \end{itemize}
\end{frame}

\begin{frame}{Por que isso importa}
  \begin{itemize}
    \item Formalidade: acesso a direitos trabalhistas, previd\^encia e prote\c{c}\~ao social \'e mediado pelo v\'inculo formal.
    \item G\^enero e ra\c{c}a/cor: diferenciais persistentes mesmo ap\'os controle por escolaridade e ocupa\c{c}\~ao s\~ao compat\'iveis com diferen\c{c}as de retorno, discrimina\c{c}\~ao e formas de segrega\c{c}\~ao ou \emph{sorting} n\~ao observadas.
    \item Roraima \'e um caso pouco estudado: economia pequena, fortemente dependente do setor p\'ublico (\PctSetorPublico\ dos ocupados), com choque migrat\'orio recente relevante para o mercado de trabalho local.
    \item Entender a composi\c{c}\~ao desses diferenciais informa pol\'iticas de formaliza\c{c}\~ao, equidade salarial e igualdade racial no estado.
  \end{itemize}
\end{frame}

\begin{frame}{Um primeiro olhar: gaps brutos (sem nenhum controle)}
  \begin{center}
    \includegraphics[width=0.95\linewidth,height=0.72\textheight,keepaspectratio]{motivacao_gaps_brutos.pdf}
  \end{center}
  \vspace{1mm}

\end{frame}

\begin{frame}{Objetivos}
  \begin{enumerate}
    \item Estimar o diferencial de rendimento associado \`a formalidade em Roraima, distinguindo a amostra ampla (todos os ocupados) da amostra restrita de empregados (prêmio salarial estrito).
    \item Estimar os diferenciais condicionais de g\^enero e das quatro categorias de ra\c{c}a/cor, com a interpreta\c{c}\~ao correta das intera\c{c}\~oes com formalidade.
    \item Isolar o papel do setor p\'ublico na estrutura salarial.
    \item Decompor exatamente o diferencial de renda mensal em efeito-pre\c{c}o (renda por hora) e efeito-jornada.
    \item Testar a estabilidade dos resultados a todos os trimestres da d\'ecada e ao per\'iodo pand\^emico.
  \end{enumerate}
\end{frame}

\section{Revis\~ao de literatura}
% =============================================================================

\begin{frame}{Base te\'orica: equa\c{c}\~ao de rendimentos}
  \begin{itemize}
    \item A especifica\c{c}\~ao segue a tradi\c{c}\~ao da \highlight{equa\c{c}\~ao de rendimentos mincer\-iana}: log do rendimento como fun\c{c}\~ao de capital humano (escolaridade, experi\^encia/idade) e caracter\'isticas do posto de trabalho.
    \begin{itemize}
      \item Mincer, J. (1974). \textit{Schooling, Experience, and Earnings}. NBER.
      \item Card, D. (1999). ``The Causal Effect of Education on Earnings''. \textit{Handbook of Labor Economics}, v.\ 3A.
    \end{itemize}
    \item A extens\~ao inclui \highlight{vari\'aveis de interesse} (formalidade, g\^enero, ra\c{c}a/cor) ao lado dos controles de capital humano -- pr\'atica padr\~ao na literatura de diferenciais salariais (Oaxaca-Blinder e extens\~oes com efeitos fixos).
  \end{itemize}
\end{frame}

\begin{frame}{Informalidade e setor p\'ublico no mercado de trabalho}
  \begin{itemize}
    \item Ulyssea, G. (2018). ``Firms, Informality, and Development: Theory and Evidence from Brazil''. \textit{American Economic Review}, 108(8).
    \begin{itemize}
      \item Modela a informalidade como resposta de firmas a custos de regula\c{c}\~ao e fiscaliza\c{c}\~ao, com implica\c{c}\~oes para produtividade e aloca\c{c}\~ao de trabalhadores.
    \end{itemize}
    \item Achado emp\'irico recorrente na literatura brasileira: trabalhadores formais recebem pr\^emio salarial mesmo ap\'os controle por escolaridade e ocupa\c{c}\~ao, consistente com segmenta\c{c}\~ao (n\~ao apenas sele\c{c}\~ao) do mercado de trabalho.
    \item Economias subnacionais com forte peso do funcionalismo p\'ublico -- caso de Roraima -- tendem a ter estrutura salarial mais comprimida no setor p\'ublico e maior premium público-privado do que a m\'edia nacional.
  \end{itemize}
\end{frame}

\begin{frame}{G\^enero e ra\c{c}a/cor no mercado de trabalho}
  \begin{itemize}
    \item Blau, F. \& Kahn, L. (2017). ``The Gender Wage Gap: Extent, Trends, and Explanations''. \textit{Journal of Economic Literature}, 55(3).
    \begin{itemize}
      \item Parte do gap de g\^enero \'e explicada por diferen\c{c}as observ\'aveis (ocupa\c{c}\~ao, jornada); uma parcela relevante permanece n\~ao explicada mesmo em especifica\c{c}\~oes ricas em controles.
    \end{itemize}
    \item Literatura brasileira sobre desigualdade racial de rendimentos documenta diferenciais persistentes entre brancos e pretos/pardos mesmo ap\'os controle por escolaridade, regi\~ao e setor formal/informal.
    \item \alert{Nota:} cita\c{c}\~oes espec\'ificas para o contexto de Roraima/regi\~ao Norte, fronteira e fluxo migrat\'orio venezuelano permanecem uma extens\~ao natural deste trabalho.
  \end{itemize}
\end{frame}

\section{Dados e metodologia}
% =============================================================================

\begin{frame}{Fonte de dados e amostra}
  \begin{itemize}
    \item \highlight{Fonte:} microdados trimestrais oficiais da PNAD Cont\'inua (IBGE), Roraima (UF~14).
    \item \highlight{Recorte principal:} 4\textordmasculine{} trimestre de cada ano, 2016T4--2025T4 -- garante comparabilidade sazonal entre os cortes e reduz a intensidade de reutiliza\c{c}\~ao das unidades amostrais do painel rotativo (Se\c{c}\~ao de Robustez testa todos os trimestres).
    \item \highlight{Funil amostral (n\~ao ponderado):}
    \begin{itemize}
      \item Total de entrevistados: \NTotal\ \quad$\to$\quad 14 anos ou mais: \NQuatorzeMais
      \item Ocupados: \NOcupados\ \quad$\to$\quad Com rendimento v\'alido: \NRendaValida
    \end{itemize}
    \item Estima\c{c}\~ao: WLS com peso amostral, dentro do desenho de r\'eplicas bootstrap oficial da PNAD Cont\'inua (200 r\'eplicas, pacote \texttt{survey} em R).
  \end{itemize}
\end{frame}

\begin{frame}{Duas amostras}
  \small
  \begin{columns}[T]
    \begin{column}{0.5\linewidth}
      \textbf{Amostra ampla (N=\NAmpla)}
      \begin{itemize}
        \item Todos os ocupados com rendimento v\'alido.
        \item \texttt{formal} pela defini\c{c}\~ao de informalidade do IBGE: posi\c{c}\~ao na ocupa\c{c}\~ao (carteira) para empregados; CNPJ para empregador/conta-pr\'opria.
        \item Usada para o \highlight{diferencial geral} de rendimento associado \`a formalidade.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\linewidth}
      \textbf{Amostra restrita (N=\NRestrita)}
      \begin{itemize}
        \item S\'o rela\c{c}\~oes de emprego: privado, dom\'estico, p\'ublico, militar/estatut\'ario.
        \item Exclui empregador, conta-pr\'opria e familiar auxiliar (32\% dos ocupados) -- onde formal/informal depende de CNPJ, n\~ao de carteira.
        \item Usada para o \highlight{pr\^emio salarial estrito} da formaliza\c{c}\~ao.
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{3mm}
  {\footnotesize Setor p\'ublico (\NSetorPublico\ postos, \PctSetorPublico\ dos ocupados) \'e identific\'avel dentro de ambas as amostras -- Se\c{c}\~ao de Resultados dedicada.}
\end{frame}

\begin{frame}{Vari\'aveis}
  \footnotesize
  \begin{columns}[T]
    \begin{column}{0.5\linewidth}
      \textbf{Dependentes}
      \begin{itemize}
        \item $\ln(\text{renda mensal real})$
        \item $\ln(\text{renda por hora real})$
      \end{itemize}
      \vspace{2mm}
      \textbf{Tratamentos}
      \begin{itemize}
        \item \texttt{formal}: ver amostras (slide anterior)
        \item \texttt{mulher}: sexo feminino
        \item \texttt{raca\_grupo}: preta, parda, ind\'igena (refer\^encia: branca; amarela exclu\'ida, N=69, 0,3\% da amostra, insuficiente para uma categoria pr\'opria)
        \item \texttt{setor\_publico}: posi\c{c}\~ao p\'ublica (com/sem carteira, militar/estatut\'ario)
      \end{itemize}
    \end{column}
    \begin{column}{0.5\linewidth}
      \textbf{Controles}
      \begin{itemize}
        \item Idade e idade\textsuperscript{2}
        \item Escolaridade (7 categorias)
        \item C\'elula ocupa\c{c}\~ao $\times$ atividade (ver pr\'oximo slide)
        \item Efeitos fixos de per\'iodo (10 trimestres)
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Especifica\c{c}\~ao econom\'etrica}
  \small
  \begin{block}{Modelo principal (WLS, desenho amostral bootstrap completo)}
    \setlength{\abovedisplayskip}{2pt}
    \setlength{\belowdisplayskip}{2pt}
    \begin{align*}
      \ln(w_{it}) = {} & \beta_1 Formal_{it} + \beta_2 Mulher_i + \beta_3 (Formal_{it}\times Mulher_i) \\
                        & + \textstyle\sum_r \beta_{4r} \text{Ra\c{c}a}_{ri} + \sum_r \beta_{5r} (Formal_{it}\times \text{Ra\c{c}a}_{ri}) + \sum_r \beta_{6r} (Mulher_i\times \text{Ra\c{c}a}_{ri}) \\
                        & + X_{it}\,\gamma + FE_{c\acute{e}lula\ ocup.\times ativ.} + FE_{per\acute{i}odo} + \varepsilon_{it}
    \end{align*}
  \end{block}
  \footnotesize
  \begin{itemize}
    \item $w_{it}$: rendimento mensal ou por hora, real (deflator oficial trimestral da PNAD Cont\'inua, espec\'ifico de UF/trimestre).
    \item $r \in \{$preta, parda, ind\'igena$\}$; c\'elula = ocupa\c{c}\~ao (11 grupos) $\times$ atividade (12 grupos), com as combina\c{c}\~oes de N$<$30 agregadas numa categoria residual ``outras'' (necess\'ario para estabilidade do desenho bootstrap -- pr\'oximo slide).
    \item Leitura causal exige cautela: a sele\c{c}\~ao para o emprego formal, ocupa\c{c}\~ao e jornada \'e potencialmente end\'ogena.
  \end{itemize}
\end{frame}

\begin{frame}{Por que c\'elula conjunta ocupa\c{c}\~ao$\times$atividade}
  \begin{itemize}
    \item Efeitos fixos separados de ocupa\c{c}\~ao e atividade controlam cada dimens\~ao \highlight{marginalmente} -- n\~ao equivalem a comparar pessoas na \highlight{mesma} combina\c{c}\~ao de ocupa\c{c}\~ao e setor.
    \item A especifica\c{c}\~ao principal usa a c\'elula conjunta (96 combina\c{c}\~oes poss\'iveis, 66 categorias finais ap\'os agregar as 31 c\'elulas com N$<$30 -- algumas com N=1 -- num bucket \'unico ``outras'', N=344, 201 UPAs).
    \item Essa agrega\c{c}\~ao n\~ao \'e s\'o parcim\^onia: sem ela, o desenho bootstrap de 200 r\'eplicas perde at\'e 74 r\'eplicas por especifica\c{c}\~ao (c\'elulas esparsas ficam vazias em boa parte das reamostragens). Com ela, \highlight{0 de 200 r\'eplicas descartadas} em toda especifica\c{c}\~ao estimada neste trabalho ($\sim$30 especifica\c{c}\~oes, incluindo todas as varia\c{c}\~oes de amostra e per\'iodo).
  \end{itemize}
\end{frame}

\begin{frame}{Interpretando as intera\c{c}\~oes com formalidade}
  \begin{itemize}
    \item Como o modelo tem \texttt{Mulher} e \texttt{Formal$\times$Mulher} simultaneamente, \highlight{o coeficiente de \texttt{Mulher} sozinho \'e o gap entre pessoas informais} -- n\~ao o gap geral.
    \item O gap entre pessoas \highlight{formais} \'e a soma \texttt{Mulher + Formal$\times$Mulher}, com erro-padr\~ao obtido da matriz de covari\^ancia completa do modelo (n\~ao da soma ingênua dos erros-padr\~ao individuais).
    \item O mesmo vale para ra\c{c}a: \texttt{Ra\c{c}a\_r} \'e o gap entre informais; \texttt{Ra\c{c}a\_r + Formal$\times$Ra\c{c}a\_r} \'e o gap entre formais.
    \item Todos os resultados a seguir j\'a reportam essa combina\c{c}\~ao correta, com teste de hip\'otese formal para a diferen\c{c}a entre os dois gaps.
  \end{itemize}
\end{frame}

    \item \alert{A amplia\c{c}\~ao do gap racial observada entre formais est\'a fortemente concentrada no \highlight{setor p\'ublico}} (17--27\%, sig.\ a 1\%) -- no privado os gaps s\~ao bem menores, e s\'o preto n\~ao \'e significativo.
    \item A diferen\c{c}a p\'ublico $-$ privado \'e significativa para preto, pardo \highlight{e} ind\'igena -- n\~ao \'e ru\'ido de uma categoria.
    \item Isso sugere que a heterogeneidade p\'ublico-privada \'e central para interpretar a intera\c{c}\~ao entre ra\c{c}a e formalidade -- compat\'ivel com mecanismos de segrega\c{c}\~ao vertical e \emph{sorting} no mercado formal documentados por Cardoso et al.\ (2025) e Gerard et al.\ (2021).
  \end{itemize}
\end{frame}

\begin{frame}{Uma pergunta complementar: quem acessa a formalidade?}
  \begin{itemize}
    \item At\'e aqui, todo o resultado responde \highlight{``quanto a formalidade est\'a associada ao rendimento''} -- o \highlight{retorno} condicional da formalidade.
    \item Mas a literatura de segmenta\c{c}\~ao/sele\c{c}\~ao do mercado de trabalho (Dalberto \& Cirino; Curi \& Menezes-Filho; Hirata \& Machado, 2010) enfatiza que a aloca\c{c}\~ao entre postos formais e informais \'e ela pr\'opria potencialmente desigual -- \highlight{acesso} \`a formalidade \'e uma dimens\~ao distinta de retorno.
    \item G\^enero e ra\c{c}a podem operar nas duas dimens\~oes ao mesmo tempo: um grupo pode ter menor probabilidade de estar em posi\c{c}\~ao formal \highlight{e}, mesmo entrando na formalidade, um pr\^emio menor.
  \end{itemize}
\end{frame}

\begin{frame}{Modelo de acesso \`a formalidade}
  \begin{itemize}
    \item Especifica\c{c}\~ao complementar e mais simples, sem intera\c{c}\~oes:
  \end{itemize}
  \small
  \begin{align*}
    P(Formal_i = 1) = {} & F(Mulher_i,\ \text{Ra\c{c}a}_i,\ Escolaridade_i,\ Idade_i, \\
                          & \phantom{F(} FE_{c\acute{e}lula\ ocup.\times ativ.},\ FE_{per\acute{i}odo})
  \end{align*}

  \begin{itemize}
    \item Logit ponderado; reportamos \highlight{efeitos marginais m\'edios} (AME, em pontos percentuais), n\~ao coeficientes em log-odds.

\section{Mecanismo: pre\c{c}o vs.\ jornada}
% =============================================================================

\begin{frame}{Decomposi\c{c}\~ao exata do diferencial mensal}
  \footnotesize
  \begin{itemize}
    \item Como $\ln(\text{renda mensal}) = \ln(\text{renda por hora}) + \ln(\text{horas semanais})$ \'e identidade alg\'ebrica, a mesma especifica\c{c}\~ao nas tr\^es vari\'aveis dependentes decomp\~oe \highlight{exatamente} (n\~ao aproximadamente) o efeito mensal em efeito-pre\c{c}o e efeito-jornada.
    \item Verificado: a soma dos coeficientes de pre\c{c}o e jornada reproduz o coeficiente mensal com diferen\c{c}a da ordem de $10^{-15}$ em todos os termos, nas duas amostras.
  \end{itemize}
  \begin{center}
    \includegraphics[width=0.78\linewidth,height=0.36\textheight,keepaspectratio]{mecanismo_jornada.pdf}
  \end{center}
  {\tiny Nota: renda por hora = renda mensal / (horas semanais $\times$ 4,33); a constante $\ln(4{,}33)$ \'e absorvida pelo intercepto e n\~ao afeta os coeficientes das vari\'aveis de interesse.}
\end{frame}

\begin{frame}{Leitura do mecanismo}
  \begin{itemize}
    \item \highlight{G\^enero:} \PctJornadaMulher{}\% do efeito mensal \'e efeito-jornada (mulheres trabalham \HorasEfeitoMulher\ horas em rela\c{c}\~ao a homens compar\'aveis, em m\'edia ponderada por ra\c{c}a); o restante \'e efeito-pre\c{c}o.
    \item \highlight{Formal$\times$Mulher:} praticamente \PctJornadaFormalMulher{}\% do efeito mensal \'e jornada -- mulheres formais trabalham jornadas mais pr\'oximas \`as de homens formais (\HorasEfeitoFormalMulher), mas o pre\c{c}o da hora entre formais n\~ao converge (slide de g\^enero).
    \item \highlight{Formal:} trabalhadores formais trabalham \emph{mais} horas (\HorasEfeitoFormal, em m\'edia ponderada por g\^enero e ra\c{c}a) e ainda ganham mais por hora -- o pr\^emio de formalidade n\~ao \'e explicado por jornada mais curta.
    \item \highlight{Ra\c{c}a/cor:} efeito majoritariamente de pre\c{c}o (73-77\% do efeito mensal \'e pre\c{c}o, 23-27\% jornada) para as tr\^es categorias.
  \end{itemize}
\end{frame}

\begin{frame}{Convertendo \% em R\$ (na renda m\'edia real da amostra)}

\begin{frame}{S\'intese -- Formalidade}
  \begin{itemize}
    \item Trabalhadores formais ganham \PctFormalGeralHoraAmpla\ a mais por hora (\ReaisFormalGeralHoraAmpla), em m\'edia (ponderado pela distribui\c{c}\~ao observada de g\^enero e ra\c{c}a) que informais compar\'aveis -- o maior diferencial estimado no estudo. Na amostra restrita de empregados, o pr\^emio salarial estrito \'e \PctFormalGeralHoraRestrita.
    \item N\~ao \'e explicado por jornada mais curta (formais trabalham \emph{mais} horas) nem \'e artefato do peso do setor p\'ublico (setor p\'ublico e formalidade s\~ao dimens\~oes distintas e conjuntamente associadas aos rendimentos).
    \item O pr\^emio varia por grupo: de \PctFormalHomemBranco\ (homens brancos) a \PctFormalMulherIndigena\ (mulheres ind\'igenas).
  \end{itemize}
\end{frame}

\begin{frame}{S\'intese -- G\^enero}
  \begin{itemize}
    \item Mulheres ganham \negativo{\PctGeneroInformal} a menos por hora (\ReaisGeneroInformal), em m\'edia (ponderado por ra\c{c}a) que homens compar\'aveis -- gap \highlight{constante} entre formais e informais (diferen\c{c}a n\~ao significativa).
    \item O gap \highlight{bruto} (sem controles) era pequeno (\BrutoGapMulherPct) -- a desigualdade s\'o aparece claramente ao comparar mulheres e homens \emph{dentro da mesma ocupa\c{c}\~ao e setor}.
    \item Parte do gap \highlight{mensal} (n\~ao do por hora) \'e jornada mais curta -- efeito misto, preponderantemente de pre\c{c}o.
    \item N\~ao encontramos evid\^encia estatisticamente significativa de diferen\c{c}a condicional no \highlight{acesso} \`a formalidade entre mulheres e homens (AME \AmePpMulherAmpla\ p.p., $p=$\PValorAmeMulherAmpla) -- IC amplo, n\~ao ``sem diferen\c{c}a''. Em contraste, o diferencial de \highlight{retorno} \'e preciso e significativo.
  \end{itemize}
\end{frame}

\begin{frame}{S\'intese -- Ra\c{c}a/cor}
  \footnotesize
  \begin{itemize}
    \item Pessoas pretas, pardas e ind\'igenas ganham entre \PctPretoInformal\ e \PctIndigenaInformal\ a menos por hora que brancas compar\'aveis, em m\'edia (ponderado por g\^enero).
    \item O gap racial \'e vis\'ivel mesmo bruto (\BrutoGapRacaPct) -- boa parte \'e desigualdade \highlight{educacional}: escolaridade reduz o gap bruto \`a metade ou mais, padr\~ao oposto ao de g\^enero.
    \item Os gaps \highlight{n\~ao s\~ao reduzidos entre formais}; sua maior magnitude est\'a fortemente concentrada no \highlight{setor p\'ublico}, onde os diferenciais condicionais alcan\c{c}am 17--27\%, frente a valores substancialmente menores no privado/dom\'estico.
    \item Efeito majoritariamente de pre\c{c}o; tend\^encia de queda ao longo do tempo, significativa para pardos.
    \item N\~ao encontramos evid\^encia significativa de diferen\c{c}a no \highlight{acesso} \`a formalidade por ra\c{c}a -- os diferenciais de \highlight{retorno}, em contraste, s\~ao precisos.
  \end{itemize}
\end{frame}

\begin{frame}{Cautelas de interpreta\c{c}\~ao}
  \begin{itemize}
    \item Estes s\~ao \highlight{diferenciais condicionais}, n\~ao efeitos causais: a sele\c{c}\~ao para emprego formal, para ocupa\c{c}\~ao/setor e para jornada permanece potencialmente end\'ogena.
    \item O gap de pessoas \highlight{pretas} (isoladamente) tem menor robustez estat\'istica em subamostras menores -- o sinal \'e consistente, a signific\^ancia n\~ao.
    \item A convers\~ao para R\$ usa a renda m\'edia da amostra como refer\^encia -- \'e ilustrativa, n\~ao uma previs\~ao individual.
    \item Pessoas amarelas (0,3\% da amostra) ficam fora da estima\c{c}\~ao de ra\c{c}a por N insuficiente para uma categoria pr\'opria.
    \item A PNAD-C n\~ao permite identificar carreira, \'org\~ao ou ente empregador no setor p\'ublico; portanto, o diferencial racial p\'ublico residual pode combinar diferen\c{c}as de remunera\c{c}\~ao dentro de posi\c{c}\~oes observ\'aveis e \emph{sorting} entre cargos/carreiras n\~ao captados pelas categorias dispon\'iveis.
  \end{itemize}
\end{frame}

\section{Conclus\~ao}

\begin{frame}{Dois resultados espelhados: mecanismos diferentes, n\~ao s\'o coeficientes diferentes}
  \begin{center}
    \footnotesize
    \renewcommand{\arraystretch}{1.5}
    \begin{tabular}{p{2.1cm}p{4.3cm}p{4.3cm}}
      \toprule
       & \highlight{G\^enero} & \highlight{Ra\c{c}a} \\
      \midrule
      N\'ivel 1: composi\c{c}\~ao & Vantagem educacional feminina $\to$ \highlight{mascara} desigualdade & Desvantagem educacional $\to$ \highlight{amplifica} desigualdade \\
      N\'ivel 2: $\times$ formalidade & Gap por hora praticamente \highlight{invari\'avel} entre formais/informais & Amplia\c{c}\~ao aparente entre formais $\to$ \highlight{setor p\'ublico} \\
      \bottomrule
    \end{tabular}
  \end{center}
  \footnotesize
  \begin{itemize}
    \item Os dois eixos s\~ao compat\'iveis com \highlight{mecanismos opostos e distintos} em cada n\'ivel -- n\~ao \'e s\'o que os coeficientes t\^em sinais/magnitudes diferentes, \'e que a \highlight{hip\'otese explicativa candidata} \'e diferente em cada caso.
  \end{itemize}
\end{frame}

\begin{frame}{S\'intese dos achados}
  \scriptsize
  \begin{itemize}
    \item \highlight{I. Formalidade} -- associada a um grande diferencial de rendimento (\PctFormalGeralHoraAmpla, em m\'edia); parte reflete composi\c{c}\~ao ocupacional/setorial (cai de M1 a M4 na sensibilidade), mas parcela substancial permanece dentro de c\'elulas compar\'aveis -- n\~ao \'e jornada nem artefato do setor p\'ublico.
    \item \highlight{II. G\^enero} -- o gap bruto praticamente inexistente (\BrutoGapMulherPct) \'e mascarado pela vantagem educacional feminina (decomposi\c{c}\~ao sequencial); condicionando composi\c{c}\~ao, emerge um gap por hora de \PctGeneroInformal\ a \PctGeneroFormal.
    \item \highlight{III. Formalidade $\times$ g\^enero} -- formalidade n\~ao reduz o gap no pre\c{c}o da hora (diferen\c{c}a n\~ao significativa), mas reduz parte da diferen\c{c}a de jornada no efeito mensal.
    \item \highlight{IV. Ra\c{c}a} -- o gap bruto (\BrutoGapRacaPct) cai \`a metade ao entrar escolaridade (desigualdade educacional amplia o gap racial, padr\~ao oposto ao de g\^enero); a amplia\c{c}\~ao do gap ``entre formais'' est\'a fortemente concentrada no \highlight{setor p\'ublico} (17--27\%, vs.\ -4,7\% a -8,0\% no privado).
    \item \highlight{Acesso \`a formalidade} vira uma qualifica\c{c}\~ao importante: n\~ao detectamos diferen\c{c}as condicionais de acesso precisas o suficiente para explicar esses padr\~oes (nenhum AME significativo a 10\%, g\^enero ou ra\c{c}a).
    \item Trabalho dom\'estico \'e a posi\c{c}\~ao com menor rendimento condicional -- ter carteira quase n\~ao altera o rendimento m\'edio (R\$\,8,39 vs.\ R\$\,8,65/hora) nem aproxima essa categoria do patamar de nenhuma outra.
    \item Achados centrais s\~ao est\'aveis a todos os trimestres da d\'ecada e ao per\'iodo pand\^emico.
  \end{itemize}
\end{frame}

\begin{frame}{Extens\~oes futuras}
  \footnotesize
  \begin{itemize}

    \item Estrat\'egias de identifica\c{c}\~ao causal (ex.: descontinuidades em regras de formaliza\c{c}\~ao) para complementar os diferenciais condicionais aqui reportados.
    \item Ampliar o modelo de acesso \`a formalidade com intera\c{c}\~oes (g\^enero$\times$ra\c{c}a) e especifica\c{c}\~oes alternativas de efeitos fixos -- a amostra atual pode ter poder estat\'istico limitado para gaps de acesso moderados.
